-- NEXUS ADMIN: member details + safe VIP cancellation -- The current project constraint allows only active/completed rentals. -- Extend it so the admin can cancel a device without deleting its history. ALTER TABLE public.nexus_machine_rentals DROP CONSTRAINT IF EXISTS nexus_machine_rentals_status_check; ALTER TABLE public.nexus_machine_rentals ADD CONSTRAINT nexus_machine_rentals_status_check CHECK (status = ANY (ARRAY['active'::text, 'completed'::text, 'cancelled'::text])); -- Run once in Supabase SQL Editor as project owner. -- This script does not delete members, change balances, or reverse ledger entries. CREATE OR REPLACE FUNCTION public.admin_get_member_details(p_user_id uuid) RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, auth AS $function$ DECLARE v_result jsonb; BEGIN IF NOT public.is_admin() THEN RAISE EXCEPTION 'admin_only'; END IF; IF p_user_id IS NULL THEN RAISE EXCEPTION 'user_id_required'; END IF; SELECT jsonb_build_object( 'member', jsonb_build_object( 'user_id', u.id, 'email', u.email, 'phone', COALESCE(NULLIF(p.phone,''), NULLIF(u.phone,'')), 'full_name', COALESCE(NULLIF(p.full_name,''), NULLIF(p.display_name,''), 'عضو بدون اسم'), 'customer_code', p.customer_code, 'member_code', p.member_code, 'referral_code', p.referral_code, 'referred_by', p.referred_by, 'vip_level', COALESCE(p.vip_level,0), 'role', p.role, 'account_status', p.account_status, 'created_at', u.created_at, 'balance', COALESCE(wb.balance,0) ), 'referrals', COALESCE(( SELECT jsonb_agg(jsonb_build_object( 'user_id', ru.id, 'full_name', COALESCE(NULLIF(rp.full_name,''), NULLIF(rp.display_name,''), 'عضو بدون اسم'), 'email', ru.email, 'phone', COALESCE(NULLIF(rp.phone,''), NULLIF(ru.phone,'')), 'customer_code', rp.customer_code, 'created_at', ru.created_at, 'deposit_count', COALESCE((SELECT count(*) FROM public.wallet_requests wr WHERE wr.user_id=ru.id AND wr.request_type='deposit'),0), 'approved_deposit_total', COALESCE((SELECT sum(wr.amount) FROM public.wallet_requests wr WHERE wr.user_id=ru.id AND wr.request_type='deposit' AND wr.status='approved'),0) ) ORDER BY ru.created_at DESC) FROM public.profiles rp JOIN auth.users ru ON ru.id=rp.id WHERE rp.referred_by=p_user_id ), '[]'::jsonb), 'rentals', COALESCE(( SELECT jsonb_agg(jsonb_build_object( 'id', r.id, 'package_id', r.package_id, 'package_name', COALESCE(mp.name, 'VIP ' || r.package_id::text), 'price', r.price, 'daily_income', r.daily_income, 'days', r.days, 'started_at', r.started_at, 'created_at', r.created_at, 'last_paid_day', r.last_paid_day, 'status', r.status ) ORDER BY r.created_at DESC) FROM public.nexus_machine_rentals r LEFT JOIN public.nexus_machine_packages mp ON mp.id=r.package_id WHERE r.user_id=p_user_id ), '[]'::jsonb), 'requests', COALESCE(( SELECT jsonb_agg(jsonb_build_object( 'id', wr.id, 'request_type', wr.request_type, 'amount', wr.amount, 'currency', wr.currency, 'request_currency', wr.request_currency, 'requested_amount', wr.requested_amount, 'status', wr.status, 'admin_note', wr.admin_note, 'created_at', wr.created_at ) ORDER BY wr.created_at DESC) FROM public.wallet_requests wr WHERE wr.user_id=p_user_id ), '[]'::jsonb), 'ledger', COALESCE(( SELECT jsonb_agg(jsonb_build_object( 'id', wl.id, 'amount', wl.amount, 'balance_after', wl.balance_after, 'entry_type', wl.entry_type, 'reason', wl.reason, 'created_at', wl.created_at ) ORDER BY wl.created_at DESC) FROM public.wallet_ledger wl WHERE wl.user_id=p_user_id ), '[]'::jsonb) ) INTO v_result FROM auth.users u LEFT JOIN public.profiles p ON p.id=u.id LEFT JOIN public.wallet_balances wb ON wb.user_id=u.id WHERE u.id=p_user_id; IF v_result IS NULL THEN RAISE EXCEPTION 'member_not_found'; END IF; RETURN v_result; END; $function$; REVOKE ALL ON FUNCTION public.admin_get_member_details(uuid) FROM PUBLIC; GRANT EXECUTE ON FUNCTION public.admin_get_member_details(uuid) TO authenticated; CREATE OR REPLACE FUNCTION public.admin_cancel_member_vip( p_rental_id bigint, p_note text DEFAULT NULL ) RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $function$ DECLARE v_user_id uuid; v_old_status text; BEGIN IF NOT public.is_admin() THEN RAISE EXCEPTION 'admin_only'; END IF; IF p_rental_id IS NULL THEN RAISE EXCEPTION 'rental_id_required'; END IF; SELECT user_id, status::text INTO v_user_id, v_old_status FROM public.nexus_machine_rentals WHERE id=p_rental_id FOR UPDATE; IF NOT FOUND THEN RAISE EXCEPTION 'vip_device_not_found'; END IF; IF lower(COALESCE(v_old_status,'')) <> 'active' THEN RAISE EXCEPTION 'vip_device_not_active'; END IF; UPDATE public.nexus_machine_rentals SET status='cancelled' WHERE id=p_rental_id; RETURN jsonb_build_object( 'ok', true, 'rental_id', p_rental_id, 'user_id', v_user_id, 'previous_status', v_old_status, 'status', 'cancelled', 'note', NULLIF(btrim(p_note),'') ); END; $function$; REVOKE ALL ON FUNCTION public.admin_cancel_member_vip(bigint,text) FROM PUBLIC; GRANT EXECUTE ON FUNCTION public.admin_cancel_member_vip(bigint,text) TO authenticated;
